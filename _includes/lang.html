{% comment %} Page language helpers, included wherever a template needs them: - nv_lang: the page language (front matter lang, default en); - t: the shared labels for that language (_data/i18n.yml); - nv_cur: this language's entry in _data/languages.yml; - nv_base: the page path without the language prefix (/sr/energy/ -> /energy/), so the same page in another language is that language's prefix + nv_base (see navigation.html and default.html). {% endcomment %} {% assign nv_lang = page.lang | default: 'en' %} {% assign t = site.data.i18n[nv_lang] | default: site.data.i18n.en %} {% assign nv_cur = site.data.languages | where: 'code', nv_lang | first %} {% assign nv_base = page.url %} {% if nv_cur and nv_cur.prefix != '' %}{% assign nv_base = page.url | remove_first: nv_cur.prefix %}{% endif %}
